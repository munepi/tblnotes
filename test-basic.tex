%% test-basic.tex
%% ==============
%% Typesetting test for every engine, mode=table (default): the
%% footnotes of tabular and tabular* are table notes below the table,
%% in the width of the table.
%%
%%   pdflatex test-basic.tex   (or xelatex / lualatex)
%%   platex  test-basic.tex && dvipdfmx test-basic
%%   uplatex -jobname=test-basic "\def\CLS{ujarticle}\input test-basic"
%% The class is article by default; set \CLS for another one (jarticle,
%% ujarticle, ltjsarticle, ...).  With pTeX the class option dvipdfmx
%% is given.
%%
%% What is covered:
%%   * a table in the body text, [t]/[c]/[b], and in a table float;
%%   * several footnotes, in l and p columns;
%%   * \footnote[<num>], \footnotemark + \footnotetext in the table;
%%   * a tabular inside a tabular (the outer one collects);
%%   * tabular* (the width of the notes is the given width);
%%   * a table without footnotes;
%%   * the counter is reset for each table; counter=<counter>;
%%     reset=false;
%%   * mode=off: \footnotemark in the table, \footnotetext after it;
%%   * footnotes of the body text are not affected.
%%
%% test-hyperref.tex runs this file with hyperref.
%%
%% What to check:
%%   * TN-RESULT OK, TN-COUNTER OK, and no error or warning.
\ifdefined\CLS\else\def\CLS{article}\fi
\ifdefined\ptexversion\def\CLSOPT{dvipdfmx}\else\def\CLSOPT{}\fi
\expandafter\documentclass\expandafter[\CLSOPT]{\CLS}
\usepackage{tablenotes}
\ifdefined\TNhyperref\usepackage{hyperref}\fi
\input{test-common}
\newcounter{tblnote}
\renewcommand\thetblnote{\fnsymbol{tblnote}}
\begin{document}

\hline
\end{tabular} between

\hline

\hline
\end{tabular} and
\begin{tabular}{|l|}
\hline
D\TNfootnote{A note of the [c] table.} \\
\hline
\end{tabular} after.

\TNcounter{footnote}{1}

7.}\TNmark{opt} & a p column\TNfootnote{A note in a p column.} \\

\footnotetext{\TNtext{fm}{table}The text for the footnotemark.}text \\

\end{tabular}

\end{table}
